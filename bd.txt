-- 1. LIMPIEZA PREVIA AUTOMÁTICA (DROP CASCADE):
--    Elimina cualquier versión anterior de tablas, vistas o tipos desactualizados
--    (soluciona errores como: column "categoria" does not exist o tipos duplicados).
-- 2. EXTENSIONES: pgcrypto y uuid-ossp.
-- 3. DOMINIOS Y ENUMS: 3 Roles oficiales (RECEPCION, TECNICO, ADMINISTRADOR) y
--    el ciclo de vida estricto de 5 fases (RECIBIDO, EN_DIAGNOSTICO, DIAGNOSTICADO, COTIZADO, ENTREGADO).
-- 4. BUCKET DE SUPABASE STORAGE:
--    Crea automáticamente el bucket público 'evidencias-radios' con sus políticas RLS
--    para subida y visualización de fotos WebP en mostrador y mesa técnica.
-- 5. CABECERA Y DETALLE MULTI-HANDY (1 a 10 equipos por orden).
-- 6. VISTAS SQL DE ALTO RENDIMIENTO para Recepción, Mesa Técnica y Administrador.
-- 7. TRIGGERS AUTOMÁTICOS para cálculo de subtotales, correlativo OS y timestamps.
-- 8. SEMILLAS (SEED DATA): Accesorios estándar, catálogo de fallas RF y usuarios demo.
-- ============================================================================

-- 1. EXTENSIONES REQUERIDAS

CREATE EXTENSION IF NOT EXISTS "pgcrypto";
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ============================================================================
-- 2. TIPOS ENUMERADOS (DOMINIOS DE NEGOCIO Y CICLO DE VIDA)
-- ============================================================================

-- Roles oficiales del sistema (Estrictamente los 3 perfiles reales del taller)
CREATE TYPE rol_usuario_enum AS ENUM (
    'RECEPCION',      -- Atención al cliente, mostrador, cotizaciones y entrega
    'TECNICO',        -- Mesa de trabajo, diagnóstico de placa y fotos internas
    'ADMINISTRADOR'   -- Acceso total, flujos, personal y configuración
);

-- Ciclo de vida estricto de una orden de servicio
CREATE TYPE estado_orden_enum AS ENUM (
    'RECIBIDO',           -- 1. Registrada por recepción con datos de cliente, handies y fotos externas
    'EN_DIAGNOSTICO',     -- 2. El técnico la abrió en su mesa de trabajo y guarda avances
    'DIAGNOSTICADO',      -- 3. El técnico concluyó fallas, fotos de placa y redacción; espera presupuesto
    'COTIZADO',           -- 4. Recepción fijó mano de obra, repuestos y generó informe A4 / Word
    'ENTREGADO',          -- 5. El cliente pagó y retiró los equipos del mostrador
    -- Estados operativos complementarios:
    'BORRADOR',
    'APROBADO',
    'RECHAZADO',
    'EN_REPARACION',
    'PARCIALMENTE_LISTO',
    'LISTO_PARA_ENTREGA',
    'ANULADO'
);

-- Estado técnico individual por equipo / handy
CREATE TYPE estado_equipo_enum AS ENUM (
    'RECIBIDO',
    'EN_DIAGNOSTICO',
    'DIAGNOSTICADO',
    'ESPERANDO_REPUESTO',
    'REPARADO',
    'RECHAZADO',
    'ENTREGADO'
);

-- Moneda oficial de liquidación
CREATE TYPE moneda_enum AS ENUM ('BOB', 'USD');

-- Bandas de telecomunicaciones RF
CREATE TYPE banda_frecuencia_enum AS ENUM ('VHF', 'UHF', 'HF', 'DUAL_BAND', 'OTRA');

-- Checklist de presencia en mostrador
CREATE TYPE presencia_accesorio_enum AS ENUM ('CON', 'SIN');

-- Condición física del accesorio o pieza
CREATE TYPE condicion_repuesto_enum AS ENUM (
    'Operativa',
    'Degradado / Baja Capacidad',
    'Dañado/Roto',
    'Se sugiere reemplazo'
);

-- Tipos de garantía técnica para informes
CREATE TYPE tipo_garantia_enum AS ENUM (
    'ESTANDAR_5_DIAS',
    'DIAS_30',
    'SIN_GARANTIA',
    'PERSONALIZADA'
);

-- ============================================================================
-- 3. FUNCIÓN TRIGGER UNIVERSAL PARA TIMESTAMP DE ACTUALIZACIÓN
-- ============================================================================
CREATE OR REPLACE FUNCTION trigger_set_timestamp()
RETURNS TRIGGER AS $$
BEGIN
    NEW.actualizado_en = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================================================================
-- 4. SUCURSALES (SOPORTE MULTI-TALLER O SUCURSAL CENTRAL)
-- ============================================================================
CREATE TABLE sucursales (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre VARCHAR(100) NOT NULL DEFAULT 'Sucursal Central - La Paz',
    ciudad VARCHAR(60) NOT NULL DEFAULT 'La Paz',
    direccion TEXT DEFAULT 'Av. 16 de Julio, Edif. América',
    telefono VARCHAR(50) DEFAULT '+591 2 2441122',
    activo BOOLEAN NOT NULL DEFAULT true,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TRIGGER trg_sucursales_timestamp
BEFORE UPDATE ON sucursales
FOR EACH ROW EXECUTE FUNCTION trigger_set_timestamp();

-- 8. CABECERA PRINCIPAL: ÓRDENES DE SERVICIO (PRESUPUESTO Y TÉRMINOS)
-- ============================================================================
CREATE TABLE ordenes_servicio (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sucursal_id UUID REFERENCES sucursales(id) ON DELETE SET NULL,
    secuencia_num BIGSERIAL,
    numero_orden VARCHAR(30) UNIQUE NOT NULL, -- Ej: OS-80412
    cliente_id UUID NOT NULL REFERENCES clientes(id) ON DELETE RESTRICT,
    
    -- Operadores involucrados
    usuario_recepcion_id UUID REFERENCES usuarios(id) ON DELETE SET NULL,
    usuario_tecnico_id UUID REFERENCES usuarios(id) ON DELETE SET NULL,
    recepcionista_nombre_manual VARCHAR(120) DEFAULT 'Recepción Mostrador',
    tecnico_nombre_manual VARCHAR(120),
    
    -- Estado global según ciclo de 5 fases
    estado estado_orden_enum NOT NULL DEFAULT 'RECIBIDO',
    moneda moneda_enum NOT NULL DEFAULT 'BOB',
    tasa_cambio NUMERIC(8, 4) NOT NULL DEFAULT 6.9600,
    
    -- Subtotales oficiales (Fase 3 y 4)
    monto_subtotal_servicio NUMERIC(12, 2) NOT NULL DEFAULT 0.00,   -- Subtotal 1: Mano de obra / calibración
    monto_subtotal_accesorios NUMERIC(12, 2) NOT NULL DEFAULT 0.00, -- Subtotal 2: Accesorios y piezas con reemplazo
    sumatoria_total NUMERIC(12, 2) NOT NULL DEFAULT 0.00,
    total_literal TEXT,                                             -- "SON: DOS MIL QUINIENTOS VEINTE 00/100 BOLIVIANOS"
    
    -- Condiciones y Parámetros Comerciales (Presupuesto e Informe A4 / Word)
    dias_vigencia INT NOT NULL DEFAULT 5,
    tiempo_entrega VARCHAR(100) NOT NULL DEFAULT '1 a 2 semanas',
    costo_revision NUMERIC(10, 2) NOT NULL DEFAULT 210.00,          -- Bs. 210,00.- / Usd. 15,00.-
    tipo_garantia VARCHAR(50) NOT NULL DEFAULT 'ESTANDAR_5_DIAS',
    texto_garantia TEXT DEFAULT '5 días calendario a partir de la entrega para verificación operativa de los trabajos realizados.',
    incluir_conclusiones BOOLEAN NOT NULL DEFAULT false,
    conclusiones JSONB DEFAULT '[]'::jsonb,
    incluir_garantia_baterias BOOLEAN NOT NULL DEFAULT true,
    referencia_adicional TEXT,
    condiciones_comerciales_json JSONB DEFAULT '{}'::jsonb,
    
    -- Trazabilidad de tiempos
    fecha_ingreso TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    fecha_compromiso DATE,
    fecha_diagnostico TIMESTAMPTZ,
    fecha_cotizacion TIMESTAMPTZ,
    fecha_entrega TIMESTAMPTZ,
    
    notas_internas TEXT,
    creado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP

    repuesto_sugerido VARCHAR(150),
    cantidad INT NOT NULL DEFAULT 1,
    precio NUMERIC(10, 2) NOT NULL DEFAULT 0.00,
    subtotal NUMERIC(10, 2) GENERATED ALWAYS AS (cantidad * precio) STORED,
    
    es_opcional BOOLEAN NOT NULL DEFAULT false,
    aprobado_cliente BOOLEAN NOT NULL DEFAULT true,
    es_manual BOOLEAN NOT NULL DEFAULT false,
    recibido_mostrador BOOLEAN NOT NULL DEFAULT false,
    orden_visual INT DEFAULT 0
);

CREATE INDEX idx_eq_repuestos_equipo ON equipo_repuestos_cotizados (equipo_id);

-- 15. BUCKET DE SUPABASE STORAGE Y POLÍTICAS DE ACCESO
-- ============================================================================
DO $$ BEGIN
    IF EXISTS (SELECT 1 FROM information_schema.schemata WHERE schema_name = 'storage') THEN
        -- 1. Crear o actualizar el bucket público 'evidencias-radios'
        INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
        VALUES (
            'evidencias-radios', 
            'evidencias-radios', 
            true, 
            10485760, -- 10 MB límite
            ARRAY['image/webp', 'image/jpeg', 'image/png', 'image/jpg']
        )
        ON CONFLICT (id) DO UPDATE SET 
            public = true,
            file_size_limit = 10485760,
            allowed_mime_types = ARRAY['image/webp', 'image/jpeg', 'image/png', 'image/jpg'];

        -- 2. Políticas de acceso para subida y lectura de imágenes
        DROP POLICY IF EXISTS "Acceso publico lectura evidencias-radios" ON storage.objects;
        CREATE POLICY "Acceso publico lectura evidencias-radios"
        ON storage.objects FOR SELECT
        USING (bucket_id = 'evidencias-radios');

        DROP POLICY IF EXISTS "Subida libre evidencias-radios" ON storage.objects;
        CREATE POLICY "Subida libre evidencias-radios"
        ON storage.objects FOR INSERT
        WITH CHECK (bucket_id = 'evidencias-radios');

        DROP POLICY IF EXISTS "Modificacion libre evidencias-radios" ON storage.objects;
        CREATE POLICY "Modificacion libre evidencias-radios"
        ON storage.objects FOR UPDATE
        USING (bucket_id = 'evidencias-radios');

        DROP POLICY IF EXISTS "Eliminacion libre evidencias-radios" ON storage.objects;
        CREATE POLICY "Eliminacion libre evidencias-radios"
        ON storage.objects FOR DELETE
        USING (bucket_id = 'evidencias-radios');
    END IF;

-- A. Vista para Recepción (/recepcion/inicio): Agrupada por las 3 columnas operativas
CREATE OR REPLACE VIEW vista_recepcion_dashboard AS
SELECT 
    o.id,
    o.numero_orden,
    c.razon_social AS cliente_nombre,
    c.telefono AS cliente_telefono,
    o.estado,
    o.moneda,
    o.sumatoria_total,
    o.monto_subtotal_servicio,
    o.monto_subtotal_accesorios,
    o.fecha_ingreso,
    (SELECT COUNT(*) FROM orden_equipos e WHERE e.orden_id = o.id) AS total_equipos,
    ROUND(EXTRACT(EPOCH FROM (NOW() - o.fecha_ingreso)) / 3600, 1) AS horas_en_taller,
    CASE 
        WHEN o.estado IN ('RECIBIDO', 'EN_DIAGNOSTICO') THEN 'EN_TALLER'
        WHEN o.estado = 'DIAGNOSTICADO' THEN 'POR_COTIZAR'
        WHEN o.estado = 'COTIZADO' THEN 'LISTAS_ENTREGA'
        ELSE 'HISTORICO'
    END AS columna_tab,
    CASE 
        WHEN o.estado IN ('RECIBIDO', 'EN_DIAGNOSTICO') THEN 'Ver Avance'
        WHEN o.estado = 'DIAGNOSTICADO' THEN 'Fijar Costos y Generar Informe'
        WHEN o.estado = 'COTIZADO' THEN 'Ver Documentos / Entregar'
        ELSE 'Ver Detalle'
    END AS accion_boton
FROM ordenes_servicio o
JOIN clientes c ON o.cliente_id = c.id
WHERE o.estado NOT IN ('ANULADO')
ORDER BY o.fecha_ingreso DESC;

-- A. Generador automático de correlativo de orden (ej. OS-80413)
CREATE OR REPLACE FUNCTION fn_generar_numero_orden()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.numero_orden IS NULL OR TRIM(NEW.numero_orden) = '' THEN
        NEW.numero_orden := 'OS-' || LPAD(NEW.secuencia_num::TEXT, 5, '0');
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generar_numero_orden
BEFORE INSERT ON ordenes_servicio
FOR EACH ROW EXECUTE FUNCTION fn_generar_numero_orden();

-- B. Recalcular automáticamente Subtotales 1, 2 y Total en ordenes_servicio
CREATE OR REPLACE FUNCTION fn_recalcular_subtotales_orden()
RETURNS TRIGGER AS $$
DECLARE
    v_orden_id UUID;
    v_subtotal1 NUMERIC(12, 2) := 0.00;
    v_subtotal2 NUMERIC(12, 2) := 0.00;
    v_total NUMERIC(12, 2) := 0.00;
BEGIN
    IF TG_TABLE_NAME = 'orden_equipos' THEN
        v_orden_id := COALESCE(NEW.orden_id, OLD.orden_id);
    ELSIF TG_TABLE_NAME = 'equipo_repuestos_cotizados' THEN
        SELECT orden_id INTO v_orden_id FROM orden_equipos WHERE id = COALESCE(NEW.equipo_id, OLD.equipo_id);
    END IF;

    IF v_orden_id IS NOT NULL THEN
        -- Subtotal 1: Servicio técnico / mano de obra (excluyendo equipos rechazados sin revisión)
        SELECT COALESCE(SUM(costo_servicio), 0.00) INTO v_subtotal1
        FROM orden_equipos
        WHERE orden_id = v_orden_id 
          AND (reparacion_rechazada IS FALSE OR no_autorizo_revision IS TRUE);

        -- Subtotal 2: Accesorios y repuestos marcados con reemplazo
        SELECT COALESCE(SUM(r.cantidad * r.precio), 0.00) INTO v_subtotal2
        FROM equipo_repuestos_cotizados r
        JOIN orden_equipos e ON r.equipo_id = e.id
        WHERE e.orden_id = v_orden_id 
          AND (e.reparacion_rechazada IS FALSE)
          AND r.se_reemplaza = true;

        v_total := v_subtotal1 + v_subtotal2;

        UPDATE ordenes_servicio
        SET monto_subtotal_servicio = v_subtotal1,
            monto_subtotal_accesorios = v_subtotal2,
            sumatoria_total = v_total,
            actualizado_en = CURRENT_TIMESTAMP
        WHERE id = v_orden_id;
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- Trigger sobre orden_equipos
CREATE TRIGGER trg_recalc_subtotales_equipos
AFTER INSERT OR UPDATE OF costo_servicio, reparacion_rechazada, no_autorizo_revision OR DELETE ON orden_equipos
FOR EACH ROW EXECUTE FUNCTION fn_recalcular_subtotales_orden();

-- Trigger sobre equipo_repuestos_cotizados
CREATE TRIGGER trg_recalc_subtotales_repuestos
AFTER INSERT OR UPDATE OF cantidad, precio, se_reemplaza OR DELETE ON equipo_repuestos_cotizados
FOR EACH ROW EXECUTE FUNCTION fn_recalcular_subtotales_orden();
